from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from ai_engine import analyze_email

app = FastAPI(title="AI Email Assistant")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)


class EmailRequest(BaseModel):
    sender: str
    subject: str
    body: str


@app.get("/")
def home():
    return {
        "message": "AI Email Assistant API is running"
    }


@app.post("/analyze")
def analyze(request: EmailRequest):
    result = analyze_email(
        sender=request.sender,
        subject=request.subject,
        body=request.body
    )

    return result
